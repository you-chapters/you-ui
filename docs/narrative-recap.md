# Narrative Recap — Frontend Implementation

Adds a `NarrativeCard` component to the dashboard that displays the free-form LLM narrative
for the current ISO week or month. Loads independently from the summary strip so the dashboard
never blocks on the LLM call.

Two backend delivery stages — the UI works identically for both:
- **Stage 1 (lazy)**: narrative generated on first page load if not cached; may take a few seconds
- **Stage 2 (scheduled)**: narrative pre-generated by cron; loads instantly from cache on page open

```ts
export interface NarrativeSummary {
  period_type: 'week' | 'month';
  period_key: string;      // "2026-W21" or "2026-05"
  entry_count: number;
  text: string;
  generated_at: string;    // ISO-8601
  is_cached: boolean;
}
```

### `ui/src/api/narrative.ts`

```ts
import type { NarrativeSummary } from '../types/narrative';
import { apiFetch } from './client';

export function getNarrative(
  type: 'week' | 'month' = 'week',
  key?: string,
  refresh = false,
): Promise<NarrativeSummary> {
  const params = new URLSearchParams({ type, refresh: String(refresh) });
  if (key) params.set('key', key);
  return apiFetch<NarrativeSummary>(`/entries/narrative?${params}`);
}
```

### `ui/src/components/NarrativeCard.tsx`

```tsx
import type { NarrativeSummary } from '../types/narrative';
import LoadingSpinner from './LoadingSpinner';

interface Props {
  narrative: NarrativeSummary | null;
  loading: boolean;
  refreshing: boolean;
  onRefresh: () => void;
}

export default function NarrativeCard({ narrative, loading, refreshing, onRefresh }: Props) {
  if (loading) return <LoadingSpinner />;
  if (!narrative) return null;

  const generatedDate = new Date(narrative.generated_at).toLocaleDateString();

  return (
    <section className="narrative-card">
      <div className="narrative-header">
        <h2>{narrative.period_type === 'week' ? 'This week' : 'This month'}</h2>
        <button onClick={onRefresh} disabled={refreshing} className="narrative-refresh">
          {refreshing ? '…' : 'Refresh'}
        </button>
      </div>
      <p className="narrative-text">{narrative.text}</p>
      <small className="narrative-meta">
        Generated {generatedDate} · {narrative.entry_count} {narrative.entry_count === 1 ? 'entry' : 'entries'}
      </small>
    </section>
  );
}
```

### `ui/src/components/NarrativeCard.test.tsx`

```tsx
import { render, screen } from '@testing-library/react';
import NarrativeCard from './NarrativeCard';
import type { NarrativeSummary } from '../types/narrative';

const stub: NarrativeSummary = {
  period_type: 'week',
  period_key: '2026-W21',
  entry_count: 3,
  text: 'It was a thoughtful week.',
  generated_at: '2026-05-20T10:00:00Z',
  is_cached: false,
};

test('renders narrative text', () => {
  render(<NarrativeCard narrative={stub} loading={false} refreshing={false} onRefresh={() => {}} />);
  expect(screen.getByText('It was a thoughtful week.')).toBeInTheDocument();
  expect(screen.getByText(/3 entries/)).toBeInTheDocument();
});

test('shows spinner when loading', () => {
  render(<NarrativeCard narrative={null} loading={true} refreshing={false} onRefresh={() => {}} />);
  // LoadingSpinner renders — narrative text not present
  expect(screen.queryByText('It was a thoughtful week.')).not.toBeInTheDocument();
});

test('disables refresh button while refreshing', () => {
  render(<NarrativeCard narrative={stub} loading={false} refreshing={true} onRefresh={() => {}} />);
  expect(screen.getByRole('button', { name: '…' })).toBeDisabled();
});
```

## Modified files

### `ui/src/pages/DashboardPage.tsx`

Add narrative state below the existing summary state. Narrative loads independently — summary
strip and people card appear first.

```tsx
import { useState, useEffect, useCallback } from 'react';
import { getSummary } from '../api/summary';
import { getNarrative } from '../api/narrative';
import type { PeriodSummary } from '../types/summary';
import type { NarrativeSummary } from '../types/narrative';
import PeriodStrip from '../components/PeriodStrip';
import PeopleCard from '../components/PeopleCard';
import NarrativeCard from '../components/NarrativeCard';
import LoadingSpinner from '../components/LoadingSpinner';
import './DashboardPage.css';

export default function DashboardPage() {
  const [period, setPeriod] = useState<7 | 30>(30);
  const [summary, setSummary] = useState<PeriodSummary | null>(null);
  const [summaryError, setSummaryError] = useState<string | null>(null);

  const [narrative, setNarrative] = useState<NarrativeSummary | null>(null);
  const [narrativeLoading, setNarrativeLoading] = useState(true);
  const [narrativeRefreshing, setNarrativeRefreshing] = useState(false);

  useEffect(() => {
    setSummary(null);
    setSummaryError(null);
    getSummary(period)
      .then(setSummary)
      .catch(e => setSummaryError(e.message ?? 'Failed to load summary'));
  }, [period]);

  useEffect(() => {
    setNarrativeLoading(true);
    getNarrative('week')
      .then(setNarrative)
      .finally(() => setNarrativeLoading(false));
  }, []);

  const handleNarrativeRefresh = useCallback(() => {
    setNarrativeRefreshing(true);
    getNarrative('week', undefined, true)
      .then(setNarrative)
      .finally(() => setNarrativeRefreshing(false));
  }, []);

  if (summaryError) return <p className="error">{summaryError}</p>;
  if (!summary) return <LoadingSpinner centered />;

  return (
    <main className="dashboard">
      <PeriodStrip summary={summary} period={period} onPeriodChange={setPeriod} />
      <PeopleCard people={summary.top_people} />
      <NarrativeCard
        narrative={narrative}
        loading={narrativeLoading}
        refreshing={narrativeRefreshing}
        onRefresh={handleNarrativeRefresh}
      />
    </main>
  );
}
```

## Verification

### Stage 1 (lazy)
1. Dashboard loads → summary strip + people card appear immediately; narrative card shows spinner
2. Narrative text appears once LLM call completes (a few seconds)
3. "Refresh" button → shows `…`, then updates text with new narrative
4. Second page load same day → narrative loads quickly from cache
5. `npm test` — NarrativeCard tests pass

### Stage 2 (scheduled — no UI changes)
6. After scheduled Lambda runs, narrative loads instantly on page open (already cached)
7. "Refresh" button still works — triggers manual regeneration via `?refresh=true`
